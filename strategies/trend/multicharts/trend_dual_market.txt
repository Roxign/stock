{ =====================================================================
  trend_dual_market ── 個股＋大盤雙趨勢（0% / 50% / 100%，只做多）
  ---------------------------------------------------------------------
  規則（與 Python 版 strategies/trend/strategy.py 的 dual_market() 相同的邏輯）：
    對「個股 (data1)」與「大盤 (data2)」各自計算：
      MA  = MALen 日簡單均線
      Sig = 近 VolLen 日「日對數報酬」樣本標準差 × √21（約一個月的波動度）
      Z   = (收盤 ÷ MA − 1) ÷ Sig
      Z > +BandK → 上升趨勢；Z < −BandK → 下降趨勢；中間維持原判定（遲滯帶）
    部位 = 50% ×（個股上升趨勢）＋ 50% ×（大盤上升趨勢）
      兩者皆多 → 100%；一多一空 → 50%；兩者皆空 → 0%

  ★ 重要近似：
    Python 版的「大盤」是 50 檔 0050 成分股的「等權重指數」（每天取有交易股票的
    平均日報酬累乘，只用當天以前的資料）。MultiCharts 單一圖表無法即時合成這個指數，
    請在同一張圖加入第二個商品當 data2，建議：
      ‧ 加權指數 (TAIEX)，或
      ‧ 元大台灣50 (0050，請用還原權值)
    兩者都是市值加權、台積電權重很高，與等權重指數的走勢會有差異，
    因此大盤訊號的轉折時點不會與 Python 回測完全相同。
    data1 與 data2 都必須是「日線」，且交易日一致。

  輸入參數：
    MALen   均線長度，預設 126（約半年）
    VolLen  波動度估計長度，預設 60
    BandK   遲滯帶寬（單位：一個月波動度），預設 0.5
    LotSize 下單單位股數：1 = 零股，1000 = 整張

  使用前請務必設定（Strategy Properties）：
    1. 日線、還原權值（除權息調整）價格。
    2. Commission：買賣各 0.1425%；證交稅：賣出 0.3%。若無法只對賣出收稅，
       可用「雙邊各 0.2925%」近似（來回 0.585%，與 Python 回測相同）。
    3. Slippage：依實際情況設定（Python 回測未計滑價，以隔日開盤價成交）。
    4. Position Limits：勾選「Allow up to 2 entry orders in the same direction
       as the currently held position ─ when the order is generated by a
       different entry order」。個股訊號與大盤訊號分別用 "Stk"、"Mkt"
       兩個進場單各買半倉。
    5. Initial Capital 例如 1,000,000；MaxBarsBack 至少 130（或 Auto-detect）。
  ===================================================================== }

inputs:
    MALen(126),
    VolLen(60),
    BandK(0.5),
    LotSize(1);

vars:
    StkMA(0), StkRet(0), StkVol(0), StkZ(0), StkUp(0),
    MktMA(0, data2), MktRet(0, data2), MktVol(0, data2), MktZ(0, data2), MktUp(0, data2),
    Eq(0), HalfQty(0);

{ ---- 個股趨勢 (data1) ---- }
StkMA = Average(Close, MALen);
if Close[1] > 0 then
    StkRet = Log(Close / Close[1]);
StkVol = StandardDev(StkRet, VolLen, 2) * SquareRoot(21);
if StkVol > 0 and StkMA > 0 then begin
    StkZ = (Close / StkMA - 1) / StkVol;
    if StkUp = 0 and StkZ > BandK then
        StkUp = 1
    else if StkUp = 1 and StkZ < -BandK then
        StkUp = 0;
end;

{ ---- 大盤趨勢 (data2) ---- }
MktMA = Average(Close of data2, MALen);
if Close[1] of data2 > 0 then
    MktRet = Log(Close of data2 / Close[1] of data2);
MktVol = StandardDev(MktRet, VolLen, 2) * SquareRoot(21);
if MktVol > 0 and MktMA > 0 then begin
    MktZ = (Close of data2 / MktMA - 1) / MktVol;
    if MktUp = 0 and MktZ > BandK then
        MktUp = 1
    else if MktUp = 1 and MktZ < -BandK then
        MktUp = 0;
end;

{ ---- 下單：兩個半倉部位各自進出 ---- }
Eq = InitialCapital + NetProfit + OpenPositionProfit;
HalfQty = IntPortion(0.5 * Eq / Close / LotSize) * LotSize;

if StkUp = 1 then begin
    if HalfQty > 0 then
        Buy ("Stk") HalfQty shares next bar at market;
end
else
    Sell ("StkX") from entry ("Stk") next bar at market;

if MktUp = 1 then begin
    if HalfQty > 0 then
        Buy ("Mkt") HalfQty shares next bar at market;
end
else
    Sell ("MktX") from entry ("Mkt") next bar at market;
